# Добавить задачи
npm run todo add "Добавить классы моделей"
npm run todo add "Добавить интерфейс для комманд"
npm run todo add "Добавить сервис, выполняющий команды"
npm run todo add "Добавить реализацию команд list, add, done и rm"

# Посмотреть список
npm run todo ls

# Отметить выполненной (ID будет 1,2,3...)
npm run todo done 1

# Посмотреть все задачи
npm run todo ls -a

# Удалить задачу
npm run todo rm 1

# Финальный список
npm run todo ls -a
